const MAX_TRADES = 2000;

function getStorageFilePath(): string {
  return (
    process.env.TURBOSCALP_AGENT_STORAGE_FILE ||
    path.resolve(process.cwd(), 'data', 'turboscalp-agent-journal.json')
  );
}

export function utcDateKey(ms = Date.now()): string {
  return new Date(ms).toISOString().slice(0, 10);
}

const store: PersistedAgentJournal = {
  version: 2,
  config: { ...DEFAULT_AGENT_CONFIG },
  dailyEquity: {
    dateUtc: utcDateKey(),
    startingEquity: 0,
    peakEquity: 0,
  },
  lastTradeAt: null,
  lastProcessedCandles: {},
  cooldowns: {},
  trades: [],
  learningSamples: [],
  logs: [],
  criticalErrors: [],
};

let loaded = false;
let loadPromise: Promise<void> | null = null;
let persistChain: Promise<void> = Promise.resolve();

export async function ensureAgentStorageLoaded(): Promise<void> {
  if (loaded) return;
  if (!loadPromise) {
    loadPromise = (async () => {
      try {
        const raw = await fs.readFile(getStorageFilePath(), 'utf8');
        const parsed = JSON.parse(raw) as Partial<PersistedAgentJournal>;
        if (parsed.config) {
          store.config = sanitizeConfig(parsed.config);
        }
        store.lastTradeAt =
          typeof parsed.lastTradeAt === 'number' ? parsed.lastTradeAt : null;
        store.lastProcessedCandles =
          parsed.lastProcessedCandles &&
          typeof parsed.lastProcessedCandles === 'object'
            ? { ...parsed.lastProcessedCandles }
            : {};
        const now = Date.now();
        store.cooldowns = {};
        if (parsed.cooldowns && typeof parsed.cooldowns === 'object') {
          for (const [sym, until] of Object.entries(parsed.cooldowns)) {
            if (typeof until === 'number' && until > now) {
              store.cooldowns[sym] = until;
            }
          }
        }
        store.trades = Array.isArray(parsed.trades)
          ? parsed.trades.slice(-MAX_TRADES)
          : [];
        store.learningSamples = Array.isArray(parsed.learningSamples)
          ? parsed.learningSamples.slice(-MAX_SAMPLES)
          : [];
        store.logs = Array.isArray(parsed.logs)
          ? parsed.logs.slice(-MAX_LOGS)
          : [];
        store.criticalErrors = Array.isArray(parsed.criticalErrors)
          ? parsed.criticalErrors.slice(-MAX_CRITICAL_ERRORS)
          : [];
        if (parsed.dailyEquity && parsed.dailyEquity.dateUtc === utcDateKey()) {
          store.dailyEquity = {
            dateUtc: parsed.dailyEquity.dateUtc,
            startingEquity: Number(parsed.dailyEquity.startingEquity) || 0,
            peakEquity: Number(parsed.dailyEquity.peakEquity) || 0,
          };
        }
      } catch (err: any) {
        if (err?.code !== 'ENOENT') {
          console.error('[agentStorage] failed to load journal:', err);
        }
      }
      loaded = true;
    })();
  }
  await loadPromise;
}

export function sanitizeConfig(
  partial: Partial<AgentEngineConfig>
): AgentEngineConfig {
  const clamp = (n: number, lo: number, hi: number) =>
    Math.min(hi, Math.max(lo, Number.isFinite(n) ? n : lo));

  return {
    environment: 'USD_M_FUTURES_DEMO',
    minSignalScore: clamp(
      Number(partial.minSignalScore ?? DEFAULT_AGENT_CONFIG.minSignalScore),
      50,
      95
    ),
    riskPerTradePct: clamp(
      Number(partial.riskPerTradePct ?? DEFAULT_AGENT_CONFIG.riskPerTradePct),
      0.05,
      DEFAULT_AGENT_CONFIG.riskPerTradePct
    ),
    maxLeverage: Math.trunc(
      clamp(
        Number(partial.maxLeverage ?? DEFAULT_AGENT_CONFIG.maxLeverage),
        1,
        DEFAULT_AGENT_CONFIG.maxLeverage
      )
    ),
    maxOpenPositions: DEFAULT_AGENT_CONFIG.maxOpenPositions,
    maxOrderNotionalUsd: clamp(
      Number(
        partial.maxOrderNotionalUsd ?? DEFAULT_AGENT_CONFIG.maxOrderNotionalUsd
      ),
      5,
      DEFAULT_AGENT_CONFIG.maxOrderNotionalUsd
    ),
    maxDailyLossPct: clamp(
      Number(partial.maxDailyLossPct ?? DEFAULT_AGENT_CONFIG.maxDailyLossPct),
      0.25,
      DEFAULT_AGENT_CONFIG.maxDailyLossPct
    ),
    cooldownMinutes: clamp(
      Number(partial.cooldownMinutes ?? DEFAULT_AGENT_CONFIG.cooldownMinutes),
      5,
      240
    ),
    requireAiApproval: true, // AI approval remains required for automated Demo entries
    localAiApprovalEnabled: partial.localAiApprovalEnabled === true,
    tradingPolicy:
      partial.tradingPolicy === 'QUANT_ONLY_PAPER' &&
      partial.quantOnlyPaperMode === true
        ? 'QUANT_ONLY_PAPER'
        : 'AI_REQUIRED',
    quantOnlyPaperMode:
      partial.tradingPolicy === 'QUANT_ONLY_PAPER' ||
      partial.quantOnlyPaperMode === true
        ? Boolean(partial.quantOnlyPaperMode ?? (partial.tradingPolicy === 'QUANT_ONLY_PAPER'))
        : false,
  };
}

export function persistAgentStorage(): Promise<void> {
  persistChain = persistChain
    .then(async () => {
      const target = getStorageFilePath();
      const dir = path.dirname(target);
      const tmp = `${target}.${Date.now()}.${Math.random().toString(36).slice(2, 6)}.tmp`;
      const payload: PersistedAgentJournal = {
        version: 2,
        config: store.config,
        dailyEquity: store.dailyEquity,
        lastTradeAt: store.lastTradeAt,
        lastProcessedCandles: store.lastProcessedCandles,
        cooldowns: store.cooldowns,
        trades: store.trades.slice(-MAX_TRADES),
        learningSamples: store.learningSamples.slice(-MAX_SAMPLES),
        logs: store.logs.slice(-MAX_LOGS),
        criticalErrors: store.criticalErrors.slice(-MAX_CRITICAL_ERRORS),
      };
      await fs.mkdir(dir, { recursive: true });
      await fs.writeFile(tmp, JSON.stringify(payload, null, 2), 'utf8');
      await fs.rename(tmp, target);
    })
    .catch((err) => {
      console.error('[agentStorage] persist failed:', err);
    });
  return persistChain;
}

export function getAgentConfig(): AgentEngineConfig {
  return { ...store.config };
}

export async function updateAgentConfig(
  partial: Partial<AgentEngineConfig>
): Promise<AgentEngineConfig> {
  await ensureAgentStorageLoaded();
  store.config = sanitizeConfig({ ...store.config, ...partial });
  await persistAgentStorage();
  return { ...store.config };
}

export function appendAgentLog(
  agent: AgentLogEntry['agent'],
  level: AgentLogEntry['level'],
  message: string,
  symbol?: string
): AgentLogEntry {
  const entry: AgentLogEntry = {
    id: `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
    at: Date.now(),
    agent,
    level,
    message,
    ...(symbol ? { symbol } : {}),
  };
  store.logs.push(entry);
  if (store.logs.length > MAX_LOGS) {
    store.logs = store.logs.slice(-MAX_LOGS);
  }
  void persistAgentStorage();
  return entry;
}

export function recordCriticalError(
  stage: string,
  message: string,
  symbol?: string
): AgentCriticalError {
  const err: AgentCriticalError = {
    id: `crit-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    at: Date.now(),
    ...(symbol ? { symbol } : {}),
    stage,
    message,
    resolved: false,
  };
  store.criticalErrors.push(err);
  if (store.criticalErrors.length > MAX_CRITICAL_ERRORS) {
    store.criticalErrors = store.criticalErrors.slice(-MAX_CRITICAL_ERRORS);
  }
  appendAgentLog('AGENT_5_EXECUTION', 'CRITICAL', `[${stage}] ${message}`, symbol);
  void persistAgentStorage();
  return err;
}

export function getCriticalErrors(limit = 50): AgentCriticalError[] {
  return store.criticalErrors
    .slice()
    .sort((a, b) => b.at - a.at)
    .slice(0, limit);
}

export function syncDailyEquity(currentEquity: number): DailyEquityState {
  const today = utcDateKey();
  if (store.dailyEquity.dateUtc !== today) {
    store.dailyEquity = {
      dateUtc: today,
      startingEquity: currentEquity > 0 ? currentEquity : 0,
      peakEquity: currentEquity > 0 ? currentEquity : 0,
    };
    void persistAgentStorage();
  } else if (currentEquity > 0) {
    if (store.dailyEquity.startingEquity <= 0) {
      store.dailyEquity.startingEquity = currentEquity;
      store.dailyEquity.peakEquity = currentEquity;
      void persistAgentStorage();
    } else if (currentEquity > store.dailyEquity.peakEquity) {
      store.dailyEquity.peakEquity = currentEquity;
      void persistAgentStorage();
    }
  }
  return { ...store.dailyEquity };
}

export function getDailyEquityState(): DailyEquityState {
  const today = utcDateKey();
  if (store.dailyEquity.dateUtc !== today) {
    return { dateUtc: today, startingEquity: 0, peakEquity: 0 };
  }
  return { ...store.dailyEquity };
}

export function getSymbolCooldown(symbol: string): number | null {
  const until = store.cooldowns[symbol];
  if (!until) return null;
  if (Date.now() >= until) {
    delete store.cooldowns[symbol];
    return null;
  }
  return until;
}

export async function startSymbolCooldown(
  symbol: string,
  minutes = store.config.cooldownMinutes
): Promise<number> {
  await ensureAgentStorageLoaded();
  const until = Date.now() + Math.max(1, minutes) * 60_000;
  store.cooldowns[symbol] = until;
  await persistAgentStorage();
  return until;
}

export async function recordTradeOpened(trade: AgentTradeRecord): Promise<void> {
  await ensureAgentStorageLoaded();
  // A retry for the same execution must not create a second journal row.
  if (store.trades.some((existing) => existing.id === trade.id)) return;
  store.lastTradeAt = trade.openedAt;
  store.trades.push(trade);
  if (store.trades.length > MAX_TRADES) {
    store.trades = store.trades.slice(-MAX_TRADES);
  }
  await persistAgentStorage();
}

export function getActiveTradeRecord(): AgentTradeRecord | null {
  // Uncertain emergency closures remain recoverable until exchange state is checked.
  const recoverable = store.trades.filter(
    (t) => t.status === 'OPEN' || t.executionStatus === 'UNCERTAIN_RECOVERY_NEEDED'
  );
  return recoverable.length > 0 ? recoverable[recoverable.length - 1] : null;
}

function isEligibleLearningSample(sample: LearningSample): boolean {
  if (!sample || sample.financialsVerified !== true) return false;
  if (typeof sample.tradeId !== 'string' || !sample.tradeId.trim()) return false;
  if (typeof sample.symbol !== 'string' || !sample.symbol.trim()) return false;
  if (sample.side !== 'BUY' && sample.side !== 'SELL') return false;
  if (!Number.isFinite(sample.netPnl) || !Number.isFinite(sample.returnPct)) return false;
  if (!Number.isFinite(sample.holdingMs) || sample.holdingMs < 0) return false;
  if (!Number.isFinite(sample.recordedAt) || sample.recordedAt <= 0) return false;
  if (!sample.features || typeof sample.features !== 'object') return false;

  const expectedOutcome: LearningSample['outcome'] =
    sample.netPnl > 1e-6 ? 'WIN' : sample.netPnl < -1e-6 ? 'LOSS' : 'FLAT';
  return sample.outcome === expectedOutcome;
}

export function getLearningSamples(limit = 200): LearningSample[] {
  // Keep the latest verified sample per trade; legacy samples without the
  // explicit marker remain in the journal but are never used for learning.
  const unique = new Map<string, LearningSample>();
  for (const sample of store.learningSamples) {
    if (isEligibleLearningSample(sample)) unique.set(sample.tradeId, sample);
  }
  const ordered = Array.from(unique.values()).sort(
    (a, b) => a.recordedAt - b.recordedAt
  );
  const safeLimit = Math.max(0, Math.trunc(Number.isFinite(limit) ? limit : 200));
  return safeLimit === 0 ? [] : ordered.slice(-safeLimit);
}

      | 'unrealizedPnl'

export async function recordTradeClosed(
  tradeId: string,
  params: {
    status: 'CLOSED' | 'EMERGENCY_CLOSED';
    exitPrice: number | null;
    realizedPnl: number;
    tradingFees: number;
    fundingFees?: number;
    closeReason: string;
    executionStatus?: ExecutionLifecycleStatus;
    financialsVerified?: boolean;
  }
): Promise<AgentTradeRecord | null> {
  await ensureAgentStorageLoaded();
  const trade = store.trades.find((t) => t.id === tradeId);
  if (!trade) return null;

  // Finalized records cannot be closed twice. An uncertain rollback stays retryable.
  if (trade.closedAt !== null && trade.executionStatus !== 'UNCERTAIN_RECOVERY_NEEDED') {
    return trade;
  }

  const executionStatus = params.executionStatus ?? 'CLOSED';
  const tradingFees = Math.abs(params.tradingFees);
  const fundingFees = params.fundingFees ?? trade.fundingFees ?? 0;
  const financiallyVerified =
    params.financialsVerified === true &&
    params.status === 'CLOSED' &&
    executionStatus === 'CLOSED' &&
    Number.isFinite(params.realizedPnl) &&
    Number.isFinite(tradingFees) &&
    Number.isFinite(fundingFees) &&
    typeof params.exitPrice === 'number' &&
    Number.isFinite(params.exitPrice) &&
    params.exitPrice > 0 &&
    Number.isFinite(trade.notionalUsd) &&
    trade.notionalUsd > 0;
  const wasEmergency =
    trade.status === 'EMERGENCY_CLOSED' ||
    trade.executionStatus === 'ROLLBACK_CLOSED' ||
    /ROLLBACK|CRITICAL_OPEN_EXPOSURE/i.test(trade.closeReason ?? '');
  const closedAt = Date.now();

  trade.status = params.status;
  trade.executionStatus = executionStatus;
  trade.exitPrice = params.exitPrice;
  trade.financialsVerified = financiallyVerified;
  trade.unrealizedPnl = 0;
  trade.closedAt = closedAt;
  trade.closeReason = params.closeReason;

  if (financiallyVerified) {
    trade.realizedPnl = params.realizedPnl;
    trade.tradingFees = tradingFees;
    trade.fundingFees = fundingFees;
    trade.commission = tradingFees;
    trade.netPnl = params.realizedPnl - tradingFees + fundingFees;
  } else {
    // Zero values supplied by rollback/fallback code are placeholders, not PnL.
    trade.realizedPnl = null;
    trade.netPnl = null;
  }

  // Emergency rollback trades are retained for audit/recovery, never used as
  // strategy-learning examples, even when their final exchange PnL is recovered.
  if (financiallyVerified && !wasEmergency) {
    const netPnl = trade.netPnl as number;
    const outcome: LearningSample['outcome'] =
      netPnl > 1e-6 ? 'WIN' : netPnl < -1e-6 ? 'LOSS' : 'FLAT';
    const returnPct = (netPnl / trade.notionalUsd) * 100;
    const sample: LearningSample = {
      id: `sample-${trade.id}`,
      tradeId: trade.id,
      symbol: trade.symbol,
      side: trade.side,
      outcome,
      netPnl,
      returnPct,
      holdingMs: Math.max(0, closedAt - trade.openedAt),
      recordedAt: closedAt,
      financialsVerified: true,
      features: trade.signalSnapshot,
    };
    const existingIndex = store.learningSamples.findIndex(
      (item) => item.tradeId === trade.id || item.id === sample.id
    );
    if (existingIndex >= 0) store.learningSamples[existingIndex] = sample;
    else store.learningSamples.push(sample);
  }

  if (store.learningSamples.length > MAX_SAMPLES) {
    store.learningSamples = store.learningSamples.slice(-MAX_SAMPLES);
  }

  await persistAgentStorage();
  return trade;
}

export function getTodayRealizedPnl(): number {

export function buildWalkForwardReport(
  samples: LearningSample[]
): WalkForwardReport {
  const ordered = samples
    .slice()
    .sort((a, b) => a.recordedAt - b.recordedAt);

  if (ordered.length < MIN_WALK_FORWARD_SAMPLES) {
    return {
      status: 'INSUFFICIENT_SAMPLES',
      minimumSamplesRequired: MIN_WALK_FORWARD_SAMPLES,
      availableSamples: ordered.length,
      retrainPolicy: 'PERIODIC_OFFLINE_ONLY',
      dynamicRiskMutation: false,
      folds: [],
      summaryNote: `Collected ${ordered.length}/${MIN_WALK_FORWARD_SAMPLES} closed trade samples required for chronological walk-forward folds. Live risk limits remain locked.`,
    };
  }

  const folds: WalkForwardFold[] = [];
  const foldCount = Math.min(3, Math.floor(ordered.length / 3));
  const testSize = Math.max(2, Math.floor(ordered.length / (foldCount + 1)));

  for (let i = 0; i < foldCount; i++) {
    const trainEnd = ordered.length - (foldCount - i) * testSize;
    const testEnd = trainEnd + testSize;
    if (trainEnd < 2) continue;

    const trainSlice = ordered.slice(0, trainEnd);
    const testSlice = ordered.slice(trainEnd, testEnd);
    if (testSlice.length === 0) continue;

    const trainWins = trainSlice.filter((s) => s.outcome === 'WIN').length;
    const testWins = testSlice.filter((s) => s.outcome === 'WIN').length;
    const testGrossWin = testSlice
      .filter((s) => s.netPnl > 0)
      .reduce((acc, s) => acc + s.netPnl, 0);
    const testGrossLoss = Math.abs(
      testSlice
        .filter((s) => s.netPnl < 0)
        .reduce((acc, s) => acc + s.netPnl, 0)
    );
    const testNetPnl = testSlice.reduce((acc, s) => acc + s.netPnl, 0);

    folds.push({
      fold: folds.length + 1,
      trainCount: trainSlice.length,
      testCount: testSlice.length,
      trainWinRate: Number(((trainWins / trainSlice.length) * 100).toFixed(1)),
      testWinRate: Number(((testWins / testSlice.length) * 100).toFixed(1)),
      testNetPnl: Number(testNetPnl.toFixed(4)),
      testProfitFactor:
        testGrossLoss > 0
          ? Number((testGrossWin / testGrossLoss).toFixed(2))
          : testGrossWin > 0
            ? 99
            : null,
    });
  }

  return {
    status: 'READY',
    minimumSamplesRequired: MIN_WALK_FORWARD_SAMPLES,
    availableSamples: ordered.length,
    retrainPolicy: 'PERIODIC_OFFLINE_ONLY',
    dynamicRiskMutation: false,
    folds,
    summaryNote: `Prepared ${folds.length} chronological walk-forward fold(s) across ${ordered.length} closed trades for periodic evaluation. Live risk limits cannot be auto-escalated.`,
  };
}

export function getPerformanceSummary(currentEquity = 0): PerformanceSummary {
  const closed = store.trades
    .filter((t) => t.status !== 'OPEN' && t.netPnl !== null)
    .slice()
    .sort((a, b) => (a.closedAt ?? a.openedAt) - (b.closedAt ?? b.openedAt));

  const openTrades = store.trades.filter((t) => t.status === 'OPEN').length;
  const wins = closed.filter((t) => (t.netPnl ?? 0) > 0).length;
  const losses = closed.filter((t) => (t.netPnl ?? 0) < 0).length;
  const grossProfit = closed
    .filter((t) => (t.netPnl ?? 0) > 0)
    .reduce((sum, t) => sum + (t.netPnl ?? 0), 0);
  const grossLoss = Math.abs(
    closed
      .filter((t) => (t.netPnl ?? 0) < 0)
      .reduce((sum, t) => sum + (t.netPnl ?? 0), 0)
  );
  const totalRealizedPnl = closed.reduce(
    (sum, t) => sum + (t.realizedPnl ?? 0),
    0
  );
  const totalTradingFees = closed.reduce(
    (sum, t) => sum + (t.tradingFees ?? t.commission ?? 0),
    0
  );
  const totalFundingFees = closed.reduce(
    (sum, t) => sum + (t.fundingFees ?? 0),
    0
  );
  const netRealizedPnl = closed.reduce((sum, t) => sum + (t.netPnl ?? 0), 0);

  let cumulative = 0;
  let peak = 0;
  let maxDrawdownUsd = 0;
  const equityCurve: PerformanceSummary['equityCurve'] = [];

  for (const t of closed) {
    cumulative += t.netPnl ?? 0;
    if (cumulative > peak) peak = cumulative;
    const dd = peak - cumulative;
    if (dd > maxDrawdownUsd) maxDrawdownUsd = dd;
    equityCurve.push({
      time: t.closedAt ?? t.openedAt,
      cumulativePnl: Number(cumulative.toFixed(4)),
      drawdownUsd: Number(dd.toFixed(4)),
    });
  }

  const daily = getDailyEquityState();
  const baseEquity =
    daily.startingEquity > 0 ? daily.startingEquity : currentEquity;
  const maxDrawdownPct =
    baseEquity > 0 ? (maxDrawdownUsd / baseEquity) * 100 : 0;
  const todayRealizedPnl = getTodayRealizedPnl();
  const dailyDrawdownPct =
    daily.startingEquity > 0 && currentEquity > 0
      ? Math.max(
          0,
          ((daily.startingEquity - currentEquity) / daily.startingEquity) * 100
        )
      : 0;

  return {
    totalTrades: store.trades.length,
    openTrades,
    closedTrades: closed.length,
    wins,
    losses,
    winRate:
      closed.length > 0
        ? Number(((wins / closed.length) * 100).toFixed(1))
        : null,
    lossRate:
      closed.length > 0
        ? Number(((losses / closed.length) * 100).toFixed(1))
        : null,
    grossProfit: Number(grossProfit.toFixed(4)),
    grossLoss: Number(grossLoss.toFixed(4)),
    profitFactor:
      grossLoss > 0
        ? Number((grossProfit / grossLoss).toFixed(2))
        : grossProfit > 0
          ? 99
          : null,
    expectancyUsd:
      closed.length > 0
        ? Number((netRealizedPnl / closed.length).toFixed(4))
        : null,
    totalRealizedPnl: Number(totalRealizedPnl.toFixed(4)),
    totalTradingFees: Number(totalTradingFees.toFixed(4)),
    totalFundingFees: Number(totalFundingFees.toFixed(4)),
    totalFees: Number(totalTradingFees.toFixed(4)),
    netRealizedPnl: Number(netRealizedPnl.toFixed(4)),
    todayRealizedPnl: Number(todayRealizedPnl.toFixed(4)),
    dailyStartingEquity: Number(daily.startingEquity.toFixed(2)),
    dailyDrawdownPct: Number(dailyDrawdownPct.toFixed(2)),
    maxDrawdownUsd: Number(maxDrawdownUsd.toFixed(4)),
    maxDrawdownPct: Number(maxDrawdownPct.toFixed(2)),
    learningSampleCount: getLearningSamples(MAX_SAMPLES).length,
    equityCurve,
    walkForwardReport: buildWalkForwardReport(getLearningSamples(MAX_SAMPLES)),
  };
}

const TELEGRAM_DEDUPE_MS = 60_000;

export async function notifyTelegramEvent(
  eventType: TelegramEventType,
  text: string,
  dedupeKey?: string
): Promise<boolean> {
  const token = (process.env.TELEGRAM_BOT_TOKEN || '').trim();
  const chatId = (process.env.TELEGRAM_CHAT_ID || '').trim();

  if (dedupeKey) {
    const prev = recentTelegramKeys.get(dedupeKey);
    if (prev && Date.now() - prev < TELEGRAM_DEDUPE_MS) {
      return false;
    }
    recentTelegramKeys.set(dedupeKey, Date.now());
  }

  try {
    const safeText = `[TURBOSCALP DEMO Â· ${eventType}]\n${text}`.slice(0, 3800);

      `https://api.telegram.org/bot${token}/sendMessage`,
      {
        chat_id: chatId,
        text: safeText,
        disable_web_page_preview: true,
      },
      { timeout: 5_000 }
    );
    return true;
  } catch (err: any) {
    // Non-blocking: Telegram failures must never interfere with exchange risk protection.
    console.warn('[telegram] notification skipped:', err?.message || 'error');

export async function reconcilePositionAndOrders(

  if (!livePos) {

    let realizedPnl = 0;
    let tradingFees = 0;
    let fundingFees = 0;
    let exitPrice: number | null = active.markPrice;
    let financialsVerified = false;
    let reconciliationError = 'Exchange income/trade history was incomplete';

    try {
      const [incomes, userTrades] = await Promise.all([

          'GET',
          '/fapi/v1/income',

          '/fapi/v1/userTrades',
          { symbol: active.symbol, startTime: active.openedAt - 5_000, limit: 50 },
          true,
          credentials
        ),
      ]);

      if (!Array.isArray(incomes) || !Array.isArray(userTrades)) {
        throw new Error('Binance returned an invalid income or user-trades response');
      }

      const validIncomeRows = incomes.filter(
        (row: any) =>
          ['REALIZED_PNL', 'COMMISSION', 'FUNDING_FEE'].includes(row?.incomeType) &&
          (typeof row.income === 'string' || typeof row.income === 'number') &&
          Number.isFinite(Number(row.income))
      );
      const realizedRows = validIncomeRows.filter(
        (row: any) => row.incomeType === 'REALIZED_PNL'
      );
      const commissionRows = validIncomeRows.filter(
        (row: any) => row.incomeType === 'COMMISSION'
      );
      const validFills = userTrades.filter(
        (row: any) =>
          (typeof row?.price === 'string' || typeof row?.price === 'number') &&
          Number.isFinite(Number(row.price)) &&
          Number(row.price) > 0
      );

      if (realizedRows.length === 0 || commissionRows.length === 0 || validFills.length === 0) {
        throw new Error('Binance history did not contain realized PnL, commission, and valid fill records');
      }

      realizedPnl = realizedRows.reduce(
        (sum: number, row: any) => sum + Number(row.income),
        0
      );
      tradingFees = commissionRows.reduce(
        (sum: number, row: any) => sum + Math.abs(Number(row.income)),
        0
      );
      fundingFees = validIncomeRows
        .filter((row: any) => row.incomeType === 'FUNDING_FEE')
        .reduce((sum: number, row: any) => sum + Number(row.income), 0);
      const lastFill = validFills[validFills.length - 1];
      exitPrice = Number(lastFill.price);

        Number.isFinite(tradingFees) &&
        Number.isFinite(fundingFees) &&
        Number.isFinite(exitPrice) &&
        (exitPrice as number) > 0;

      }
    } catch (err: any) {

    }

    // Cancel any leftover bot-owned protective Algo orders cleanly
    await cancelAlgoOrderSafe(
      binanceRequest,
      credentials,
      active.symbol,
      active.stopAlgoId,
      active.stopClientAlgoId
    );
    await cancelAlgoOrderSafe(
      binanceRequest,
      credentials,
      active.symbol,
      active.targetAlgoId,
      active.targetClientAlgoId
    );

    if (!financialsVerified) {

        activeTradeStillOpen: true,
        reconcileNote: `${active.symbol} is flat on exchange; financial history is unverified. Recovery remains pending.`,
      };
    }

    const closed = await recordTradeClosed(active.id, {
      status: 'CLOSED',
      exitPrice,
      realizedPnl,
      tradingFees,
      fundingFees,
      closeReason: 'EXCHANGE_FLAT_RECONCILED',
      executionStatus: 'CLOSED',
      financialsVerified: true,
    });

    await startSymbolCooldown(active.symbol, cooldownMinutes);

    const netPnl = closed?.netPnl ?? realizedPnl - tradingFees + fundingFees;
    appendAgentLog(
      'AGENT_6_MONITOR',
      'TRADE',
      `${active.symbol} closed on exchange Â· realized=${realizedPnl.toFixed(4)} USDT Â· fees=${tradingFees.toFixed(4)} Â· funding=${fundingFees.toFixed(4)} Â· net=${netPnl.toFixed(4)} USDT Â· ${cooldownMinutes}m cooldown started`,
      active.symbol
    );

    void notifyTelegramEvent(
      'POSITION_CLOSED',
      `${active.symbol} ${active.side} closed @ $${(exitPrice ?? active.entryPrice).toFixed(4)}\nNet PnL: ${netPnl >= 0 ? '+' : ''}${netPnl.toFixed(4)} USDT (Realized ${realizedPnl.toFixed(4)}, Fees ${tradingFees.toFixed(4)}, Funding ${fundingFees.toFixed(4)})`
    );

    return {
      snapshot,
      activeTradeStillOpen: false,
      reconcileNote: `Closed ${active.symbol} reconciled (net ${netPnl >= 0 ? '+' : ''}$${netPnl.toFixed(2)})`,
    };
  }

  // Still open: reconcile mark price, unrealized PnL, and protective Algo Order status
  const matchingAlgos = snapshot.algoOrders.filter(
    (o: any) => o.symbol === active.symbol
  );
  const slAlive = matchingAlgos.some(
    (o: any) =>
      o.clientAlgoId === active.stopClientAlgoId ||
      (active.stopAlgoId && String(o.algoId) === String(active.stopAlgoId))
  );
  const tpAlive = matchingAlgos.some(
    (o: any) =>
      o.clientAlgoId === active.targetClientAlgoId ||
      (active.targetAlgoId && String(o.algoId) === String(active.targetAlgoId))
  );

  // Also sample accrued funding fees while open
  let accruedFunding = active.fundingFees ?? 0;
  try {
    const fundingRows = await binanceRequest<any[]>(
      'GET',
      '/fapi/v1/income',
      {
        symbol: active.symbol,
        incomeType: 'FUNDING_FEE',
        startTime: active.openedAt,
        limit: 50,
      },
      true,
      credentials
    );
    if (Array.isArray(fundingRows)) {
      accruedFunding = fundingRows.reduce((s, r) => s + toNum(r.income), 0);
    }
  } catch {
    // Non-fatal
  }

  if (!slAlive) {
    recordCriticalError(
      'MONITOR_MISSING_SL',
      `Protective STOP_MARKET Algo order missing for open ${active.symbol} position.`,
      active.symbol
    );
  }

  await updateOpenTradeMetrics(active.id, {
    markPrice: livePos.markPrice,
    entryPrice: livePos.entryPrice > 0 ? livePos.entryPrice : active.entryPrice,
    executedQuantity: String(Math.abs(livePos.positionAmt)),
    unrealizedPnl: livePos.unRealizedProfit,
    fundingFees: accruedFunding,
    stopStatus: slAlive ? 'ARMED' : 'FAILED',
    takeProfitStatus: tpAlive ? 'ARMED' : 'FAILED',
  });

  return {
    snapshot,
    activeTradeStillOpen: true,
    reconcileNote: `${active.symbol} OPEN Â· entry=$${livePos.entryPrice.toFixed(4)} mark=$${livePos.markPrice.toFixed(4)} uPnL=${livePos.unRealizedProfit >= 0 ? '+' : ''}$${livePos.unRealizedProfit.toFixed(2)}`,
  };

const toNum = (v: unknown): number => {
  const n = Number(v);
  return Number.isFinite(n) ? n : 0;
};

export function extractFillMetrics(
  orderResp: any,
  fallbackPrice: number
): {
  orderId: number | string | null;
  status: string;
  executedQty: number;
  avgPrice: number;
} {
  const executedQty = toNum(orderResp?.executedQty);
  const directAvg = toNum(orderResp?.avgPrice);
  const cumQuote = toNum(orderResp?.cumQuote ?? orderResp?.cumQuoteQty);
  const derivedAvg =
    directAvg > 0
      ? directAvg
      : executedQty > 0 && cumQuote > 0
        ? cumQuote / executedQty
        : fallbackPrice;

  return {
    orderId: orderResp?.orderId ?? null,
    status: String(orderResp?.status || 'UNKNOWN'),
    executedQty,
    avgPrice: derivedAvg,
  };
}

export function buildProtectiveAlgoParams(params: {
  symbol: string;
  entrySide: 'BUY' | 'SELL';
  positionMode: 'ONE_WAY' | 'HEDGE';
  positionSide: 'BOTH' | 'LONG' | 'SHORT';
  type: 'STOP_MARKET' | 'TAKE_PROFIT_MARKET';
  triggerPrice: string;
  clientAlgoId: string;
}): Record<string, string> {
  const closeSide = params.entrySide === 'BUY' ? 'SELL' : 'BUY';
  // Binance rule: with closePosition=true, do NOT send quantity or reduceOnly.
  // Use triggerPrice (not deprecated stopPrice) and workingType=MARK_PRICE.
  return {
    algoType: 'CONDITIONAL',
    symbol: params.symbol,
    side: closeSide,
    positionSide: params.positionMode === 'HEDGE' ? params.positionSide : 'BOTH',
    type: params.type,
    triggerPrice: params.triggerPrice,
    workingType: 'MARK_PRICE',
    closePosition: 'true',
    clientAlgoId: params.clientAlgoId,
  };
}

export async function fetchOpenAlgoOrdersSafe(
  binanceRequest: BinanceRequestFn,
  credentials: DemoCredentials,

      '/fapi/v1/algoOpenOrders',

export async function cancelAlgoOrderSafe(
  binanceRequest: BinanceRequestFn,
  credentials: DemoCredentials,
  symbol: string,
  algoId?: number | string | null,
  clientAlgoId?: string
): Promise<boolean> {
  if (!algoId && !clientAlgoId) return false;
  try {
    const params: Record<string, any> = { symbol };
    if (algoId) params.algoId = algoId;
    else if (clientAlgoId) params.clientAlgoId = clientAlgoId;
    await binanceRequest('DELETE', '/fapi/v1/algoOrder', params, true, credentials);
    return true;
  } catch {
    return false;
  }
}

async function detectPositionMode(

      true,
      credentials
    );
    return dual?.dualSidePosition === true ? 'HEDGE' : 'ONE_WAY';
  } catch {

  }

export async function executeApprovedOrder(
  decision: RiskDecision,
  quant: QuantAnalysisResult,
  aiResult: GeminiAnalysisResult,
  filter: ExchangeSymbolFilter,
  ctx: ExecutionContext
): Promise<ExecutionOutcome> {
  const plan = decision.orderPlan;
  if (!decision.approved || !plan) {
    return {
      success: false,
      tradeRecord: null,
      error: 'Execution rejected: missing Risk Manager approval.',
    };
  }

  if (
    ctx.inFlightSymbols.has(plan.symbol) ||
    hasClientOrderId(plan.clientOrderId)
  ) {
    return {
      success: false,
      tradeRecord: null,
      error: `Duplicate execution blocked for ${plan.symbol} (${plan.clientOrderId}).`,
    };
  }

  ctx.inFlightSymbols.add(plan.symbol);
  const tradeId = `tr-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
  const stopClientAlgoId = `SL_${tradeId.slice(3, 15)}`;
  const targetClientAlgoId = `TP_${tradeId.slice(3, 15)}`;

  try {
    const positionMode = await detectPositionMode(
      ctx.binanceRequest,
      ctx.credentials
    );
    const positionSide: 'BOTH' | 'LONG' | 'SHORT' =
      positionMode === 'HEDGE'
        ? plan.side === 'BUY'
          ? 'LONG'
          : 'SHORT'
        : 'BOTH';

    // Enforce leverage <= 2x on Binance Demo before entry
    await ctx.binanceRequest(
      'POST',
      '/fapi/v1/leverage',
      { symbol: plan.symbol, leverage: plan.leverage },
      true,
      ctx.credentials
    );

    const entryParams: Record<string, any> = {
      symbol: plan.symbol,
      side: plan.side,
      type: 'MARKET',
      quantity: plan.quantity,
      newClientOrderId: plan.clientOrderId,
      newOrderRespType: 'RESULT',
    };
    if (positionMode === 'HEDGE') {
      entryParams.positionSide = positionSide;
    }

    let entryResp: any;
    try {
      entryResp = await ctx.binanceRequest<any>(
        'POST',
        '/fapi/v1/order',
        entryParams,
        true,
        ctx.credentials
      );
    } catch (submitErr: any) {
      // Do not blindly retry an ambiguous market order: check clientOrderId status on Binance first
      try {
        entryResp = await ctx.binanceRequest<any>(
          'GET',
          '/fapi/v1/order',
          { symbol: plan.symbol, origClientOrderId: plan.clientOrderId },
          true,
          ctx.credentials
        );
        appendAgentLog(
          'AGENT_5_EXECUTION',
          'WARN',
          `Recovered ambiguous entry order ${plan.clientOrderId} via exchange query.`,
          plan.symbol
        );
      } catch {
        throw submitErr;
      }
    }

    const fill = extractFillMetrics(entryResp, plan.entryPrice);
    if (fill.executedQty <= 0) {
      throw new Error(
        `Entry order ${plan.clientOrderId} returned executedQty=0 (status=${fill.status}).`
      );
    }

    const actualQtyStr = floorToStep(fill.executedQty, filter.stepSize);
    const actualNumericQty = Number(actualQtyStr) || fill.executedQty;
    const isPartial =
      Math.abs(actualNumericQty - plan.numericQuantity) > filter.stepSize * 0.5;
    const actualNotional = Number((actualNumericQty * fill.avgPrice).toFixed(4));

    // Recalculate protective trigger prices relative to actual fill price while preserving stop/TP distance
    const slDist = Math.abs(plan.entryPrice - Number(plan.stopPrice));
    const tpDist = Math.abs(Number(plan.takeProfitPrice) - plan.entryPrice);
    const adjustedStop = roundPriceToTick(
      plan.side === 'BUY' ? fill.avgPrice - slDist : fill.avgPrice + slDist,
      filter.tickSize
    );
    const adjustedTp = roundPriceToTick(
      plan.side === 'BUY' ? fill.avgPrice + tpDist : fill.avgPrice - tpDist,
      filter.tickSize
    );

    const stopParams = buildProtectiveAlgoParams({
      symbol: plan.symbol,
      entrySide: plan.side,
      positionMode,
      positionSide,
      type: 'STOP_MARKET',
      triggerPrice: adjustedStop,
      clientAlgoId: stopClientAlgoId,
    });

    const tpParams = buildProtectiveAlgoParams({
      symbol: plan.symbol,
      entrySide: plan.side,
      positionMode,
      positionSide,
      type: 'TAKE_PROFIT_MARKET',
      triggerPrice: adjustedTp,
      clientAlgoId: targetClientAlgoId,
    });

    let stopAlgoId: number | string | null = null;
    let targetAlgoId: number | string | null = null;
    let protectionError = '';

    try {
      const stopResp = await ctx.binanceRequest<any>(
        'POST',
        '/fapi/v1/algoOrder',
        stopParams,
        true,
        ctx.credentials
      );
      stopAlgoId = stopResp?.algoId ?? null;

      const tpResp = await ctx.binanceRequest<any>(
        'POST',
        '/fapi/v1/algoOrder',
        tpParams,
        true,
        ctx.credentials
      );
      targetAlgoId = tpResp?.algoId ?? null;

      // Verify BOTH protective orders are genuinely open via GET /fapi/v1/openAlgoOrders
      const openAlgos = await fetchOpenAlgoOrdersSafe(
        ctx.binanceRequest,
        ctx.credentials,
        plan.symbol
      );
      const slVerified = openAlgos.some(
        (o) =>
          o.clientAlgoId === stopClientAlgoId ||
          (stopAlgoId && String(o.algoId) === String(stopAlgoId))
      );
      const tpVerified = openAlgos.some(
        (o) =>
          o.clientAlgoId === targetClientAlgoId ||
          (targetAlgoId && String(o.algoId) === String(targetAlgoId))
      );

      if (!slVerified || !tpVerified) {
        protectionError = `Protective exit verification failed on openAlgoOrders (SL=${slVerified}, TP=${tpVerified}).`;
      }
    } catch (algoErr: any) {
      protectionError = algoErr?.message || 'Algo protective order rejected.';
    }

    const baseRecord: AgentTradeRecord = {
      id: tradeId,
      symbol: plan.symbol,
      side: plan.side,
      positionMode,
      positionSide,
      status: 'OPEN',
      executionStatus: isPartial ? 'PARTIAL_PROTECTED' : 'FILLED_PROTECTED',
      requestedQuantity: plan.quantity,
      executedQuantity: actualQtyStr,
      leverage: plan.leverage,
      notionalUsd: actualNotional,
      riskUsd: Number((actualNumericQty * slDist).toFixed(4)),
      entryPrice: fill.avgPrice,
      exitPrice: null,
      markPrice: fill.avgPrice,
      stopPrice: Number(adjustedStop),
      takeProfitPrice: Number(adjustedTp),
      entryOrderId: fill.orderId,
      entryClientOrderId: plan.clientOrderId,
      stopAlgoId,
      stopClientAlgoId,
      targetAlgoId,
      targetClientAlgoId,
      emergencyOrderId: null,
      stopStatus: protectionError ? 'FAILED' : 'ARMED',
      takeProfitStatus: protectionError ? 'FAILED' : 'ARMED',
      unrealizedPnl: 0,
      realizedPnl: null,
      tradingFees: 0,
      fundingFees: 0,
      commission: 0,
      netPnl: null,
      openedAt: Date.now(),
      closedAt: null,
      closeReason: null,
      signalSnapshot: {
        quantDirection: quant.direction,
        heuristicScore: quant.heuristicScore,
        trendStrength: quant.trendStrength,
        rsi14: quant.rsi14,
        ema9: quant.ema9_15m,
        ema21: quant.ema21_15m,
        ema50: quant.ema50_15m,
        atr14: quant.atr14,
        volatilityPct: quant.volatilityPct,
        volumeRatio: quant.volumeRatio,
        breakout: quant.breakout,
        mtfAligned: quant.mtfAligned,
        marketRegime: aiResult.marketRegime,
        aiDirection: aiResult.direction,
        aiSentiment: aiResult.sentiment,
        aiConfidence: aiResult.confidence,
        aiRiskFlags: aiResult.riskFlags,
        stopDistancePct: plan.stopDistancePct,
      },
    };

    if (protectionError) {
      // Protective Exit Failure Protocol:
      // 1. Cancel whichever bot-owned Algo protective order succeeded
      await cancelAlgoOrderSafe(
        ctx.binanceRequest,
        ctx.credentials,
        plan.symbol,
        stopAlgoId,
        stopClientAlgoId
      );
      await cancelAlgoOrderSafe(
        ctx.binanceRequest,
        ctx.credentials,
        plan.symbol,
        targetAlgoId,
        targetClientAlgoId
      );

      // 2. Safely close actual remaining position via MARKET
      const closeSide = plan.side === 'BUY' ? 'SELL' : 'BUY';
      const emergencyClientId = `EMG_${tradeId.slice(3, 15)}`;
      const closeParams: Record<string, any> = {
        symbol: plan.symbol,
        side: closeSide,
        type: 'MARKET',
        quantity: actualQtyStr,
        newClientOrderId: emergencyClientId,
        newOrderRespType: 'RESULT',
      };
      if (positionMode === 'HEDGE') {
        closeParams.positionSide = positionSide;
      } else {
        closeParams.reduceOnly = 'true';
      }

      let emergencyOrderId: number | string | null = null;
      let verifiedFlat = false;
      try {
        const closeResp = await ctx.binanceRequest<any>(
          'POST',
          '/fapi/v1/order',
          closeParams,
          true,
          ctx.credentials
        );
        emergencyOrderId = closeResp?.orderId ?? null;
        // 3. Verify resulting position state on exchange
        const posRisk = await ctx.binanceRequest<any[]>(
          'GET',
          '/fapi/v3/positionRisk',
          { symbol: plan.symbol },
          true,
          ctx.credentials
        );
        const remaining = (Array.isArray(posRisk) ? posRisk : []).find(
          (p) =>
            p.symbol === plan.symbol &&
            (positionMode === 'ONE_WAY' || p.positionSide === positionSide)
        );
        verifiedFlat = Math.abs(toNum(remaining?.positionAmt)) <= 1e-12;
      } catch (closeErr: any) {
        protectionError += ` | Emergency close failed: ${closeErr?.message || 'unknown'}`;
      }

      baseRecord.emergencyOrderId = emergencyOrderId;
      baseRecord.status = 'EMERGENCY_CLOSED';
      baseRecord.executionStatus = verifiedFlat
        ? 'ROLLBACK_CLOSED'
        : 'UNCERTAIN_RECOVERY_NEEDED';

      await recordTradeOpened(baseRecord);
      await recordTradeClosed(baseRecord.id, {
        status: 'EMERGENCY_CLOSED',
        exitPrice: fill.avgPrice,
        realizedPnl: 0,
        tradingFees: 0,
        closeReason: verifiedFlat
          ? `ROLLBACK_FLAT: ${protectionError}`
          : `CRITICAL_OPEN_EXPOSURE: ${protectionError}`,
        executionStatus: baseRecord.executionStatus,
      });

      // 4. Pause new automated entries + 5. Create CRITICAL activity alert & Telegram
      ctx.onPauseRequired(
        `Protective exit failed on ${plan.symbol}; automated entries paused`
      );
      recordCriticalError(
        'PROTECTIVE_EXIT_ROLLBACK',
        `${protectionError} (verifiedFlat=${verifiedFlat})`,
        plan.symbol
      );
      void notifyTelegramEvent(
        'CRITICAL_FAILURE',
        `CRITICAL protective exit failure on ${plan.symbol}: ${protectionError}. Emergency close flat=${verifiedFlat}. New entries PAUSED.`
      );

      return {
        success: false,
        tradeRecord: baseRecord,
        error: protectionError,
      };
    }

    await recordTradeOpened(baseRecord);

    appendAgentLog(
      'AGENT_5_EXECUTION',
      'TRADE',
      `DEMO ENTRY ${plan.side} ${actualQtyStr} ${plan.symbol} @ $${fill.avgPrice.toFixed(4)} (${plan.leverage}x) | SL Algo #${stopAlgoId} ($${adjustedStop}) | TP Algo #${targetAlgoId} ($${adjustedTp})${isPartial ? ' [PARTIAL FILL]' : ''}`,
      plan.symbol
    );

    void notifyTelegramEvent(
      'DEMO_ORDER_EXECUTED',
      `EXECUTED ${plan.side} ${actualQtyStr} ${plan.symbol} @ $${fill.avgPrice.toFixed(4)} ($${actualNotional.toFixed(2)}, ${plan.leverage}x)`
    );
    void notifyTelegramEvent(
      'SL_CONFIRMED',
      `SL ARMED ${plan.symbol}: Algo #${stopAlgoId} trigger=$${adjustedStop} (MARK_PRICE)`
    );
    void notifyTelegramEvent(
      'TP_CONFIRMED',
      `TP ARMED ${plan.symbol}: Algo #${targetAlgoId} trigger=$${adjustedTp} (MARK_PRICE)`
    );

    return {
      success: true,
      tradeRecord: baseRecord,
    };
  } catch (err: any) {
    const msg = err?.message || 'Execution error';
    recordCriticalError('ENTRY_EXECUTION', msg, plan.symbol);
    void notifyTelegramEvent(
      'CRITICAL_FAILURE',
      `Execution failed for ${plan.symbol}: ${msg}`
    );
    return {
      success: false,
      tradeRecord: null,
      error: msg,
    };
  } finally {
    ctx.inFlightSymbols.delete(plan.symbol);
  }
}

function makeCandles(
  count: number,
  intervalMs: number,
  startPrice: number,
  step: number,
  endMs: number,
  spikeLastVol = false
): RawKline[] {
  const out: RawKline[] = [];
  const startOpen = endMs - count * intervalMs;
  for (let i = 0; i < count; i++) {
    const openTime = startOpen + i * intervalMs;
    const closeTime = openTime + intervalMs - 1;
    const open = Math.max(1, startPrice + i * step);
    const close = Math.max(1, open + step * 0.92);
    const high = Math.max(open, close) + Math.abs(step) * 0.5 + 0.1;
    const low = Math.max(0.5, Math.min(open, close) - Math.abs(step) * 0.5 - 0.1);
    const isLast = i === count - 1;
    out.push({
      openTime,
      open,
      high,
      low,
      close: isLast ? close + (step >= 0 ? 1 : -1) * Math.abs(step) * 2.5 : close,
      volume: isLast && spikeLastVol ? 9000 : 1200 + (i % 4) * 50,
      closeTime,
    });
  }
  return out;
}

  // math qty = 5 / 500 = 0.01 BTC -> notional $500 -> capped at MAX_ORDER_NOTIONAL $100 = 0.002 BTC
  const smallCapInput = makeRiskInput(now);
  smallCapInput.account.availableBalance = 2_000;
  const approved = evaluateRisk(smallCapInput);
  assert.equal(approved.approved, true);
  assert.equal(approved.orderPlan!.quantity, '0.002');
  assert.ok(approved.orderPlan!.notionalUsd <= HARD_RISK_CONSTANTS.MAX_ORDER_NOTIONAL_USD);
  assert.ok(approved.orderPlan!.riskPercentOfAvailable <= HARD_RISK_CONSTANTS.RISK_PER_TRADE_PCT);
  assert.equal(floorToStep(0.00299, 0.001), '0.002');
  assert.equal(roundPriceToTick(49500.14, 0.1), '49500.1');

  // When risk budget is smaller than $100 notional cap: balance = $10_000, sl distance = 5% ($2500/unit)
  // riskAmount = $25 -> qty = 25 / 2500 = 0.01 BTC? Wait 0.01*50000=$500 capped; try balance=$2000, sl=48000 ($2000/unit):
  // riskAmount = $5 -> qty = 5/2000 = 0.0025 -> step floored 0.002 -> risk = $4.00 (no leverage multiplication)
  const riskBounded = makeRiskInput(now);
  riskBounded.account.availableBalance = 2_000;
  riskBounded.signal.stopLoss = 48_000;
  riskBounded.signal.takeProfit = 53_000;
  const rbRes = evaluateRisk(riskBounded);
  assert.equal(rbRes.approved, true);
  assert.equal(rbRes.orderPlan!.quantity, '0.002');

  // All mandatory rejection conditions:
  const cases: Array<[string, (i: RiskEvaluationInput) => void]> = [
    ['disconnected', (i) => (i.account.connected = false)],
    ['permission disabled', (i) => (i.account.canTrade = false)],
    ['order test unconfirmed', (i) => (i.account.orderTestPassed = false)],
    ['stale data', (i) => (i.signal.fetchedAt = now - 120_000)],
    ['low score', (i) => (i.signal.signalStrengthScore = 60)],
    ['AI DEGRADED', (i) => { i.aiAnalysis!.available = false; i.aiAnalysis!.degraded = true; }],
    ['AI conflict', (i) => (i.aiAnalysis!.direction = 'SELL')],
    ['invalid stop-loss', (i) => (i.signal.stopLoss = i.signal.price + 100)],
    ['daily loss limit', (i) => (i.dailyEquity.realizedPnlToday = -105)],
    ['open position', (i) => (i.openPositions = [{ symbol: 'ETHUSDT', positionAmt: 0.1 }])],
    ['duplicate order', (i) => (i.duplicateClientOrderExists = true)],
    ['cooldown', (i) => (i.cooldownUntilMs = now + 600_000)],
  ];

  for (const [label, mutate] of cases) {
    const scenario = makeRiskInput(now);
    mutate(scenario);
    const out = evaluateRisk(scenario);
    assert.equal(out.approved, false, `Expected rejection for: ${label}`);
  }
});

    if (method === 'DELETE' && endpoint === '/fapi/v1/algoOrder') {
      canceledAlgo = true;
      return { success: true };
    }
    if (method === 'POST' && endpoint === '/fapi/v1/order' && params?.side === 'SELL') {
      marketCloseCalled = true;
      return { orderId: 556, status: 'FILLED', executedQty: '0.002' };
    }
    if (method === 'GET' && endpoint === '/fapi/v3/positionRisk') {
      return [{ symbol: 'BTCUSDT', positionAmt: '0' }];
    }
    return {};
  }) as BinanceRequestFn;

  const quantMock = analyzeMarketQuantitatively(

      symbol: 'BTCUSDT',

  assert.equal(canceledAlgo, true);

  assert.equal(rollbackOutcome.tradeRecord?.financialsVerified, false);
  assert.equal(rollbackOutcome.tradeRecord?.realizedPnl, null);
  assert.equal(rollbackOutcome.tradeRecord?.netPnl, null);

    getLearningSamples(20).some(
      (sample) => sample.tradeId === rollbackOutcome.tradeRecord?.id
    ),

  );
